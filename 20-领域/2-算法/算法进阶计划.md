### A1. 二分查找

```java
// 标准二分：查 target，找不到返回 -1（闭区间 [l, r]）
int binarySearch(int[] nums, int target) {
    int l = 0, r = nums.length - 1;
    while (l <= r) {
        int mid = l + (r - l) / 2;               // 防溢出，别写 (l+r)/2
        if (nums[mid] == target) return mid;
        else if (nums[mid] < target) l = mid + 1;
        else r = mid - 1;
    }
    return -1;
}
// 左边界：第一个 >= target 的位置（半开区间 [l, r)，找不到返回 nums.length）
int leftBound(int[] nums, int target) {
    int l = 0, r = nums.length;
    while (l < r) {
        int mid = l + (r - l) / 2;
        if (nums[mid] >= target) r = mid;        // 左边界：偏左收缩
        else l = mid + 1;
    }
    return l;
}
```

### A2. 滑动窗口

```java
// 最长无重复子串骨架
int lengthOfLongestSubstring(String s) {
    Map<Character, Integer> map = new HashMap<>();   // 字符 -> 出现次数
    int l = 0, ans = 0;
    for (int r = 0; r < s.length(); r++) {
        char c = s.charAt(r);
        map.put(c, map.getOrDefault(c, 0) + 1);      // 右指针扩窗
        while (map.get(c) > 1) {                     // 条件不满足，收缩左边
            char left = s.charAt(l);
            map.put(left, map.get(left) - 1);
            l++;
        }
        ans = Math.max(ans, r - l + 1);              // 窗口有效，更新答案
    }
    return ans;
}
```

### A3. 矩阵 BFS / DFS

```java
int[][] dirs = {{0,1},{0,-1},{1,0},{-1,0}};

// BFS：最短步数（入队即标记！防重复入队）
int bfs(char[][] grid) {
    int m = grid.length, n = grid[0].length;
    boolean[][] visited = new boolean[m][n];
    Deque<int[]> q = new ArrayDeque<>();
    q.offer(new int[]{0, 0});
    visited[0][0] = true;
    int steps = 0;
    while (!q.isEmpty()) {
        for (int i = q.size(); i > 0; i--) {         // 按层遍历才能数步数
            int[] cur = q.poll();
            for (int[] d : dirs) {
                int nx = cur[0]+d[0], ny = cur[1]+d[1];
                if (nx<0 || nx>=m || ny<0 || ny>=n) continue;   // 越界
                if (visited[nx][ny] || grid[nx][ny]=='0') continue;
                visited[nx][ny] = true;
                q.offer(new int[]{nx, ny});
            }
        }
        steps++;
    }
    return steps;
}

// DFS：岛屿数量（原地标记省空间）
int numIslands(char[][] grid) {
    int count = 0;
    for (int i = 0; i < grid.length; i++)
        for (int j = 0; j < grid[0].length; j++)
            if (grid[i][j] == '1') { count++; dfs(grid, i, j); }
    return count;
}
void dfs(char[][] grid, int i, int j) {
    if (i<0 || i>=grid.length || j<0 || j>=grid[0].length) return;
    if (grid[i][j] != '1') return;
    grid[i][j] = '0';                                 // 标记访问
    dfs(grid, i-1, j); dfs(grid, i+1, j);
    dfs(grid, i, j-1); dfs(grid, i, j+1);
}
```

### A4. DP 一维 + 背包

```java
// 打家劫舍：不能偷相邻两家
int rob(int[] nums) {
    int n = nums.length;
    if (n == 0) return 0;
    if (n == 1) return nums[0];                       // 边界：单元素
    int[] dp = new int[n];
    dp[0] = nums[0];
    dp[1] = Math.max(nums[0], nums[1]);
    for (int i = 2; i < n; i++)
        dp[i] = Math.max(dp[i-1], dp[i-2] + nums[i]); // 不偷 or 偷
    return dp[n-1];
}

// 01背包：每个物品只能用一次（内层倒序）
int knapsack(int[] weight, int[] value, int W) {
    int[] dp = new int[W + 1];
    for (int i = 0; i < weight.length; i++)
        for (int w = W; w >= weight[i]; w--)
            dp[w] = Math.max(dp[w], dp[w - weight[i]] + value[i]);
    return dp[W];
}
// 完全背包（无限次用）：内层正序 for (int w = weight[i]; w <= W; w++)
```

### A5. 单调栈

// 右侧第一个更大元素（栈里存下标）
int[] nextGreater(int[] nums) {
    int n = nums.length;
    int[] res = new int[n];
    Deque<Integer> stack = new ArrayDeque<>();
    for (int i = n - 1; i >= 0; i--) {
        while (!stack.isEmpty() && nums[stack.peek()] <= nums[i])
            stack.pop();
        res[i] = stack.isEmpty() ? -1 : nums[stack.peek()];

ListNode reverse(ListNode head) {       // 反转链表
    ListNode prev = null, cur = head;
    while (cur != null) {
        ListNode next = cur.next;       // ① 先存后继，否则断链
        cur.next = prev;                // ② 掉头
        prev = cur;                     // ③ 前移
        cur = next;
    }
    return prev;                        // 新头是 prev
}

// 快慢指针：判环 / 找中点（同一骨架）
boolean hasCycle(ListNode head) {
    ListNode slow = head, fast = head;
    while (fast != null && fast.next != null) {
        slow = slow.next;
        fast = fast.next.next;
        if (slow == fast) return true;
    }
    return false;
}

ListNode mergeTwoLists(ListNode l1, ListNode l2) {   // 哨兵节点合并
    ListNode dummy = new ListNode(0);   // 哨兵：免去头节点特判
    ListNode tail = dummy;
    while (l1 != null && l2 != null) {
        if (l1.val < l2.val) { tail.next = l1; l1 = l1.next; }
        else                 { tail.next = l2; l2 = l2.next; }
        tail = tail.next;
    }
    tail.next = l1 != null ? l1 : l2;
    return dummy.next;                  // 返回 dummy.next
}

List<List<Integer>> levelOrder(TreeNode root) {
    List<List<Integer>> res = new ArrayList<>();
    if (root == null) return res;
    Deque<TreeNode> q = new ArrayDeque<>();
    q.offer(root);
    while (!q.isEmpty()) {
        List<Integer> level = new ArrayList<>();
        for (int i = q.size(); i > 0; i--) {
            TreeNode node = q.poll();
            level.add(node.val);
            if (node.left != null) q.offer(node.left);
            if (node.right != null) q.offer(node.right);
        }
        res.add(level);
    }
    return res;
}

// BST 合法性校验（必须传 min/max，用 long 防溢出）
boolean isValidBST(TreeNode root) { return check(root, Long.MIN_VALUE, Long.MAX_VALUE); }
boolean check(TreeNode node, long min, long max) {
    if (node == null) return true;
    if (node.val <= min || node.val >= max) return false;  // 严格开区间
    return check(node.left, min, node.val) && check(node.right, node.val, max);
}

// 最近公共祖先 LCA
TreeNode lowestCommonAncestor(TreeNode root, TreeNode p, TreeNode q) {
    if (root == null || root == p || root == q) return root;
    TreeNode left = lowestCommonAncestor(root.left, p, q);
    TreeNode right = lowestCommonAncestor(root.right, p, q);
    if (left != null && right != null) return root;
    return left != null ? left : right;
}

// 拓扑排序 Kahn（"课程表"系列）
int[] topoSort(int n, int[][] prerequisites) {
    List<List<Integer>> graph = new ArrayList<>();
    int[] indegree = new int[n];
    for (int i = 0; i < n; i++) graph.add(new ArrayList<>());
    for (int[] p : prerequisites) {
        graph.get(p[1]).add(p[0]);   // 先学 p[1] 才能学 p[0]
        indegree[p[0]]++;
    }
    Deque<Integer> q = new ArrayDeque<>();
    for (int i = 0; i < n; i++)
        if (indegree[i] == 0) q.offer(i);
    List<Integer> order = new ArrayList<>();
    while (!q.isEmpty()) {
        int u = q.poll();
        order.add(u);
        for (int v : graph.get(u))
            if (--indegree[v] == 0) q.offer(v);
    }
    // order.size() == n 无环；不等则存在循环依赖
    return order.stream().mapToInt(i -> i).toArray();
}

// 并查集：使用前 for (int i = 0; i < n; i++) parent[i] = i;
int[] parent;
int find(int x) {
    if (parent[x] != x)
        parent[x] = find(parent[x]);   // 路径压缩
    return parent[x];
}
void union(int x, int y) {
    int rx = find(x), ry = find(y);
    if (rx != ry) parent[rx] = ry;
}
boolean connected(int x, int y) { return find(x) == find(y); }
