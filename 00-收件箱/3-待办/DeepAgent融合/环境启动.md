# 第 2 步：安装 Spring Boot Starter

# 默认模式（意图识别场景演示）
mvn -pl deepagent-spring-boot-example spring-boot:run

# 其他模式（通过环境变量切换）
$env:DEMO_TODO_INTERRUPTION_ENABLED = "true"      # Todo 中断/恢复演示
$env:DEMO_CUSTOM_ASK_USER_ENABLED = "true"        # 自定义 AskUser 演示
$env:PPT_ENABLED = "true"                         # PPT 动态生成演示
mvn -pl deepagent-spring-boot-example spring-boot:run
